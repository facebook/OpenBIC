17/*
18 * MP29526 - Digital Multi-Phase Controller (MPS)
19 *
20 * Datasheet : MP29526 Rev. 0.1, 2/26/2026 (preliminary)
21 * Register map : MP29526/MP29529/MP29426 Register Map Rev 1.0.0.196+ /
22 * 2.0.0.233+ - one PMBus map shared by the whole family,
23 * distinguished only by IC_DEVICE_ID (ADh) and by how many
24 * rails are physically present on a given part.
25 * Reference : mp29816a.c / mp29816a.h
26 *
27 * This file is written against the shared family register map, not against
28 * any one project's rail count. Command codes, widths, bit fields and the
29 * per-rail page arithmetic below apply to MP29426/MP29529/MP29526 alike;
30 * mp29526_check_device_id() accepts an IC_DEVICE_ID from any of the three.
31 * How many of the up to 4 rails a specific board actually wires up (and
32 * which sensor/page each rail maps to) is a per-project fact that belongs
33 * in the platform layer (cfg->arg0 / vr_pre_read_args), not in this driver.
34 *
35 * PAGE MAP (PAGE 00h bits[5:0]):
36 * 0 = Rail 1 operating registers 5 = MPS registers affecting Rail 1
37 * 1 = Rail 2 operating registers 6 = MPS registers affecting Rail 2
38 * 2 = Rail 3 operating registers 7 = MPS registers affecting Rail 3
39 * 3 = Rail 4 operating registers 8 = MPS registers affecting Rail 4
40 * 4 = SVID psys 9,10 = MPS all-rail, 11 = debug
41 *
42 * Rail N (0-indexed, MP29526_RAIL_1..4) selects Page N / Page (5+N); a part
43 * with fewer physical rails simply never has the platform ask for the
44 * pages beyond its rail count.
45 */

65/*
66 * VOUT_MODE (20h) encodings. Every VID-format register - VOUT_COMMAND,
67 * VOUT_MAX/MIN, VOUT_OV/UV_FAULT_LIMIT, VBOOT, READ_VOUT - is scaled by
68 * whichever of these the rail is configured for.
69 */
70#define MP29526_VOUT_MODE_DIRECT_1MV 0x40 /* 1 mV/LSB */
71#define MP29526_VOUT_MODE_VID_10MV 0x22 /* VID table, 10 mV/LSB */
72#define MP29526_VOUT_MODE_VID_5MV 0x21 /* VID table, 5 mV/LSB */
73#define MP29526_VOUT_MODE_LINEAR_512 0x17 /* linear, 1/512 V = 1.953125mV */
74
75/* WRITE_PROTECT (10h) */
76#define MP29526_WP_DISABLE 0x00
77#define MP29526_WP_ALL_EXCEPT_WP 0x80
78#define MP29526_WP_EXCEPT_WP_OP_PAGE 0x40
79#define MP29526_WP_EXCEPT_WP_OP_PAGE_ONOFF_VOUT 0x20

81/* STATUS_MFR_PROTECT (80h on the rail's MFR page, 5-8), 2 bytes, read only */

84#define MP29526_PROTECT_VAUX_FAULT BIT(2)
